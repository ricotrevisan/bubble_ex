-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS "custom__Onboarding Answer" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "label" TEXT,
  "score" REAL,
  PRIMARY KEY ("_id"),
  FOREIGN KEY ("Created By") REFERENCES "custom__User" ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__Survey Response" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "answer" TEXT,
  "onboarding answer" TEXT,
  "rating" REAL,
  "status" TEXT,
  PRIMARY KEY ("_id"),
  FOREIGN KEY ("Created By") REFERENCES "custom__User" ("_id"),
  FOREIGN KEY ("onboarding answer") REFERENCES "custom__Onboarding Answer" ("_id"),
  FOREIGN KEY ("status") REFERENCES "option__Status Type" ("db_value")
);

CREATE TABLE IF NOT EXISTS "custom__User" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "email" TEXT,
  PRIMARY KEY ("_id"),
  FOREIGN KEY ("Created By") REFERENCES "custom__User" ("_id")
);

CREATE TABLE IF NOT EXISTS "option__Status Type" (
  "db_value" TEXT NOT NULL,
  "Display" TEXT,

  PRIMARY KEY ("db_value")
