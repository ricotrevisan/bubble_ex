CREATE SCHEMA IF NOT EXISTS "custom";

CREATE TABLE "custom"."Draft" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."Note" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Body" text,
  "Secret" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."Public page" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Slug text" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."User" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "email" text,
  "Admin" boolean,
  PRIMARY KEY ("_id")
);

ALTER TABLE "custom"."Draft" ADD FOREIGN KEY ("Created By") REFERENCES "custom"."User" ("_id");
ALTER TABLE "custom"."Note" ADD FOREIGN KEY ("Created By") REFERENCES "custom"."User" ("_id");
ALTER TABLE "custom"."Public page" ADD FOREIGN KEY ("Created By") REFERENCES "custom"."User" ("_id");
ALTER TABLE "custom"."User" ADD FOREIGN KEY ("Created By") REFERENCES "custom"."User" ("_id");
