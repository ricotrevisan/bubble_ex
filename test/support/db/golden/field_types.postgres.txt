CREATE SCHEMA IF NOT EXISTS "custom";
CREATE SCHEMA IF NOT EXISTS "option";

CREATE TABLE "custom"."Project" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Name" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."Task" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Attachment" text,
  "Budget" jsonb,
  "Checks" boolean[],
  "Cover" text,
  "Dates" timestamptz[],
  "Done" boolean,
  "Due" timestamptz,
  "Duration" jsonb,
  "Estimate" double precision,
  "Files" text[],
  "Images" text[],
  "Labels" text[],
  "Notes" text[],
  "Owner" text,
  "Place" jsonb,
  "Places" jsonb[],
  "Project" text,
  "Ranges" jsonb[],
  "Scores" double precision[],
  "Status" text,
  "Subtasks" text[],
  "Title" text,
  "Watchers" text[],
  "Window" jsonb,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."User" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "email" text,
  "Nickname" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "option"."Status" (
  "db_value" text,
  "Display" text,
  "Color" text,
  PRIMARY KEY ("db_value")
);
