CREATE SCHEMA IF NOT EXISTS "custom";

  "Files" text[],

  "Labels" text[],
  "Notes" text[],
  "Owner" text,
  "Place" jsonb,
  "Places" jsonb[],

  "Ranges" jsonb[],

  "Status" text,
  "Subtasks" text[],
  "Title" text,
  "Watchers" text[],

CREATE TABLE "custom"."User" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "email" text,
  "Nickname" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "option"."Status" (
  "db_value" text,
  "Display" text,

);
