-- SQLite DDL for Bubble app
-- NOTE: arrays stored as JSON text; FKs require PRAGMA foreign_keys = ON.
-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS "option__Channel" (
  "Display" TEXT NOT NULL,
  "db_value" TEXT,
  "sort_factor" REAL,
  PRIMARY KEY ("Display")
);

CREATE TABLE IF NOT EXISTS "option__Legacy" (
  "Display" TEXT NOT NULL,
  PRIMARY KEY ("Display")
);

CREATE TABLE IF NOT EXISTS "option__Orphan" (
  "Display" TEXT NOT NULL,
  PRIMARY KEY ("Display")
);

CREATE TABLE IF NOT EXISTS "option__Priority" (
  "Display" TEXT NOT NULL,
  "channels" TEXT,
  "color" TEXT,
  "db_value" TEXT,
  "deleted" TEXT,
  "escalates_to" TEXT,
  "icon" TEXT,
  "sort_factor" REAL,
  "tooltip" TEXT,
  "urgent" TEXT,
  "weight" REAL,
  PRIMARY KEY ("Display")
);

CREATE TABLE IF NOT EXISTS "custom__Ticket" (
  "Channels" TEXT,  -- list<TEXT>; store as JSON
  "Priority" TEXT,
  "_id" TEXT NOT NULL,
  PRIMARY KEY ("_id"),
  FOREIGN KEY ("Priority") REFERENCES "option__Priority" ("Display")
);
