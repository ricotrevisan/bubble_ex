-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

CREATE TABLE IF NOT EXISTS "custom__Ticket" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Channels" TEXT,  -- list<TEXT>; store as JSON
  "Priority" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__User" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "email" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "option__Channel" (
  "db_value" TEXT NOT NULL,
  "Display" TEXT,
  PRIMARY KEY ("db_value")
);

CREATE TABLE IF NOT EXISTS "option__Orphan" (
  "db_value" TEXT NOT NULL,

  PRIMARY KEY ("db_value")
);

CREATE TABLE IF NOT EXISTS "option__Priority" (

  "Urgent" INTEGER,

  PRIMARY KEY ("db_value")
);
