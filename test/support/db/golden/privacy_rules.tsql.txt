CREATE SCHEMA [custom];
GO

CREATE TABLE [custom].[Draft] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  CONSTRAINT [PK_draft] PRIMARY KEY ([_id])
);

CREATE TABLE [custom].[Note] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [Body] NVARCHAR(MAX),
  [Secret] NVARCHAR(MAX),
  CONSTRAINT [PK_note] PRIMARY KEY ([_id])
);

CREATE TABLE [custom].[Public page] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [Slug text] NVARCHAR(MAX),
  CONSTRAINT [PK_public_page] PRIMARY KEY ([_id])
);

CREATE TABLE [custom].[User] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [email] NVARCHAR(MAX),
  [Admin] BIT,
  CONSTRAINT [PK_user] PRIMARY KEY ([_id])
);

-- References without a foreign key (Bubble does not enforce referential integrity):
-- [custom].[Draft].[Created By] -> [custom].[User].[_id]
-- [custom].[Note].[Created By] -> [custom].[User].[_id]
-- [custom].[Public page].[Created By] -> [custom].[User].[_id]
-- [custom].[User].[Created By] -> [custom].[User].[_id]
