import { z } from 'zod';

// Project
export const ProjectSchema = z.object({
  Name: z.string().nullish(),
  _id: z.string(),
});

export type Project = z.infer<typeof ProjectSchema>;

// Status
export const StatusSchema = z.object({
  Display: z.string(),
  color: z.string().nullish(),
  db_value: z.string().nullish(),
  sort_factor: z.number().nullish(),
});

export type Status = z.infer<typeof StatusSchema>;

// Task
export const TaskSchema = z.object({
  Attachment: z.string().nullish(),
  Budget: z.object({}).passthrough().nullish(), // bubble_number_range (structured Bubble type)
  Checks: z.array(z.boolean()).nullish(),
  Cover: z.string().nullish(),
  Dates: z.array(z.string().datetime()).nullish(),
  Done: z.boolean().nullish(),
  Due: z.string().datetime().nullish(),
  Duration: z.object({}).passthrough().nullish(), // bubble_dateinterval (structured Bubble type)
  Estimate: z.number().nullish(),
  Files: z.array(z.string()).nullish(),
  Images: z.array(z.string()).nullish(),
  Labels: z.array(z.string()).nullish(), // enum -> status option set (values not in IR)
  Notes: z.array(z.string()).nullish(),
  Owner: z.string().nullish(), // reference -> user
  Place: z.object({}).passthrough().nullish(), // bubble_geo_address (structured Bubble type)
  Places: z.array(z.object({}).passthrough()).nullish(), // bubble_geo_address (structured Bubble type)
  Project: z.string().nullish(), // reference -> project
  Ranges: z.array(z.object({}).passthrough()).nullish(), // bubble_date_range (structured Bubble type)
  Scores: z.array(z.number()).nullish(),
  Status: z.string().nullish(), // enum -> status option set (values not in IR)
  Subtasks: z.array(z.string()).nullish(), // reference -> task
  Title: z.string().nullish(),
  Watchers: z.array(z.string()).nullish(), // reference -> user
  Window: z.object({}).passthrough().nullish(), // bubble_date_range (structured Bubble type)
  _id: z.string(),
});

export type Task = z.infer<typeof TaskSchema>;

// User
export const UserSchema = z.object({
  Nickname: z.string().nullish(),
  _id: z.string(),
});

export type User = z.infer<typeof UserSchema>;
