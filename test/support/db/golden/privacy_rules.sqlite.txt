-- SQLite DDL for Bubble app
-- NOTE: arrays stored as JSON text.
-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

CREATE TABLE IF NOT EXISTS "custom__Draft" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__Note" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Body" TEXT,
  "Secret" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__Public page" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Slug text" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__User" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "email" TEXT,
  "Admin" INTEGER,
  PRIMARY KEY ("_id")
);

-- References without a foreign key (Bubble does not enforce referential integrity):
-- "custom__Draft"."Created By" -> "custom__User"."_id"
-- "custom__Note"."Created By" -> "custom__User"."_id"
-- "custom__Public page"."Created By" -> "custom__User"."_id"
-- "custom__User"."Created By" -> "custom__User"."_id"
