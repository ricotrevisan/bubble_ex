CREATE SCHEMA IF NOT EXISTS "custom";

CREATE TYPE "Address" AS (
  "Latitude" double precision,
  "Lines" text[],
  "Seen" bigint,
  "Street" text,
  "Verified" boolean
);
CREATE TYPE "B" AS (
  "A" jsonb,
  "blob" jsonb
);
CREATE TYPE "A" AS (
  "At" timestamptz,
  "B" "B"
);
CREATE TYPE "Node" AS (
  "Children" jsonb,
  "Label" text
);

CREATE TABLE "custom"."Shipment" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Address" "Address",
  "Blank" jsonb,
  "Invalid" jsonb,
  "Invalid list" jsonb,
  "Missing" jsonb,
  "Nowhere" jsonb,
  "Pings" "A"[],
  "Tree" "Node",
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."User" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "email" text,
  PRIMARY KEY ("_id")
);
