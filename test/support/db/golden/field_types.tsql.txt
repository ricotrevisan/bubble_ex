CREATE SCHEMA [option];

CREATE TABLE [custom].[Project] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [Name] NVARCHAR(MAX),
  CONSTRAINT [PK_project] PRIMARY KEY ([_id])
);

CREATE TABLE [custom].[Task] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [Attachment] NVARCHAR(MAX),
  [Budget] NVARCHAR(MAX),
  [Checks] NVARCHAR(MAX) /* list<bit>: consider a junction table */,
  [Cover] NVARCHAR(MAX),
  [Dates] NVARCHAR(MAX) /* list<datetime>: consider a junction table */,
  [Done] BIT,
  [Due] DATETIME2,
  [Duration] NVARCHAR(MAX),
  [Estimate] FLOAT,
  [Files] NVARCHAR(MAX) /* list<text>: consider a junction table */,
  [Images] NVARCHAR(MAX) /* list<text>: consider a junction table */,
  [Labels] NVARCHAR(MAX) /* list<enum>: consider a junction table */,
  [Notes] NVARCHAR(MAX) /* list<text>: consider a junction table */,
  [Owner] NVARCHAR(450),
  [Place] NVARCHAR(MAX),
  [Places] NVARCHAR(MAX) /* list<text>: consider a junction table */,
  [Project] NVARCHAR(450),
  [Ranges] NVARCHAR(MAX) /* list<text>: consider a junction table */,
  [Scores] NVARCHAR(MAX) /* list<float>: consider a junction table */,
  [Status] NVARCHAR(450),
  [Subtasks] NVARCHAR(MAX) /* list<ref>: consider a junction table */,
  [Title] NVARCHAR(MAX),
  [Watchers] NVARCHAR(MAX) /* list<ref>: consider a junction table */,
  [Window] NVARCHAR(MAX),
  CONSTRAINT [PK_task] PRIMARY KEY ([_id])
);

CREATE TABLE [custom].[User] (
  [_id] NVARCHAR(450),
  [Created Date] DATETIME2,
  [Modified Date] DATETIME2,
  [Created By] NVARCHAR(450),
  [Slug] NVARCHAR(MAX),
  [email] NVARCHAR(MAX),
  [Nickname] NVARCHAR(MAX),
  CONSTRAINT [PK_user] PRIMARY KEY ([_id])
);

CREATE TABLE [option].[Status] (
  [db_value] NVARCHAR(450),
  [Display] NVARCHAR(MAX),
  [Color] NVARCHAR(MAX),
  CONSTRAINT [PK_status] PRIMARY KEY ([db_value])
);
