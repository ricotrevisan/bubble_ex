-- SQLite DDL for Bubble app
-- NOTE: arrays stored as JSON text; FKs require PRAGMA foreign_keys = ON.
-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS "custom__Project" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Name" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__Task" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Attachment" TEXT,
  "Budget" TEXT,
  "Checks" TEXT,  -- list<INTEGER>; store as JSON
  "Cover" TEXT,
  "Dates" TEXT,  -- list<TEXT>; store as JSON
  "Done" INTEGER,
  "Due" TEXT,
  "Duration" TEXT,
  "Estimate" REAL,
  "Files" TEXT,  -- list<TEXT>; store as JSON
  "Images" TEXT,  -- list<TEXT>; store as JSON
  "Labels" TEXT,  -- list<TEXT>; store as JSON
  "Notes" TEXT,  -- list<TEXT>; store as JSON
  "Owner" TEXT,
  "Place" TEXT,
  "Places" TEXT,  -- list<TEXT>; store as JSON
  "Project" TEXT,
  "Ranges" TEXT,  -- list<TEXT>; store as JSON
  "Scores" TEXT,  -- list<REAL>; store as JSON
  "Status" TEXT,
  "Subtasks" TEXT,  -- list<TEXT>; store as JSON
  "Title" TEXT,
  "Watchers" TEXT,  -- list<TEXT>; store as JSON
  "Window" TEXT,
  PRIMARY KEY ("_id"),
  FOREIGN KEY ("Owner") REFERENCES "custom__User" ("_id"),
  FOREIGN KEY ("Project") REFERENCES "custom__Project" ("_id"),
  FOREIGN KEY ("Status") REFERENCES "option__Status" ("db_value")
);

CREATE TABLE IF NOT EXISTS "custom__User" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "email" TEXT,
  "Nickname" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "option__Status" (
  "db_value" TEXT NOT NULL,
  "Display" TEXT,
  "Color" TEXT,
  PRIMARY KEY ("db_value")
);
