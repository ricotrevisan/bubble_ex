-- SQLite DDL for Bubble app
-- NOTE: arrays stored as JSON text.
-- NOTE: Bubble groups (custom/option/api) have no SQLite schema equivalent; encoded as a table-name prefix.

CREATE TABLE IF NOT EXISTS "custom__Project" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Name" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__Task" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "Attachment" TEXT,
  "Budget" TEXT,
  "Checks" TEXT,  -- list<INTEGER>; store as JSON
  "Cover" TEXT,
  "Dates" TEXT,  -- list<TEXT>; store as JSON
  "Done" INTEGER,
  "Due" TEXT,
  "Duration" TEXT,
  "Estimate" REAL,
  "Files" TEXT,  -- list<TEXT>; store as JSON
  "Images" TEXT,  -- list<TEXT>; store as JSON
  "Labels" TEXT,  -- list<TEXT>; store as JSON
  "Notes" TEXT,  -- list<TEXT>; store as JSON
  "Owner" TEXT,
  "Place" TEXT,
  "Places" TEXT,  -- list<TEXT>; store as JSON
  "Project" TEXT,
  "Ranges" TEXT,  -- list<TEXT>; store as JSON
  "Scores" TEXT,  -- list<REAL>; store as JSON
  "Status" TEXT,
  "Subtasks" TEXT,  -- list<TEXT>; store as JSON
  "Title" TEXT,
  "Watchers" TEXT,  -- list<TEXT>; store as JSON
  "Window" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "custom__User" (
  "_id" TEXT NOT NULL,
  "Created Date" TEXT,
  "Modified Date" TEXT,
  "Created By" TEXT,
  "Slug" TEXT,
  "email" TEXT,
  "Nickname" TEXT,
  PRIMARY KEY ("_id")
);

CREATE TABLE IF NOT EXISTS "option__Status" (
  "db_value" TEXT NOT NULL,
  "Display" TEXT,
  "Color" TEXT,
  PRIMARY KEY ("db_value")
);

-- References without a foreign key (Bubble does not enforce referential integrity):
-- "custom__Project"."Created By" -> "custom__User"."_id"
-- "custom__Task"."Created By" -> "custom__User"."_id"
-- "custom__Task"."Owner" -> "custom__User"."_id"
-- "custom__Task"."Project" -> "custom__Project"."_id"
-- "custom__Task"."Status" -> "option__Status"."db_value"
-- "custom__User"."Created By" -> "custom__User"."_id"
