CREATE SCHEMA IF NOT EXISTS "custom";
CREATE SCHEMA IF NOT EXISTS "option";

CREATE TABLE "custom"."Ticket" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "Channels" text[],
  "Priority" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "custom"."User" (
  "_id" text,
  "Created Date" timestamptz,
  "Modified Date" timestamptz,
  "Created By" text,
  "Slug" text,
  "email" text,
  PRIMARY KEY ("_id")
);

CREATE TABLE "option"."Channel" (
  "db_value" text,
  "Display" text,
  PRIMARY KEY ("db_value")
);

CREATE TABLE "option"."Orphan" (
  "db_value" text,
  "Display" text,
  "Parent" text,
  PRIMARY KEY ("db_value")
);

CREATE TABLE "option"."Priority" (
  "db_value" text,
  "Display" text,
  "Channels" text[],
  "Color" text,
  "Escalates to" text,
  "Icon" text,
  "Urgent" boolean,
  "Weight" double precision,
  PRIMARY KEY ("db_value")
);
