import { z } from 'zod';

// User
export const UserSchema = z.object({
  _id: z.string(),
  'Created Date': z.string().datetime().nullish(),
  'Modified Date': z.string().datetime().nullish(),
  'Created By': z.string().nullish(), // reference -> user
  Slug: z.string().nullish(),
  email: z.string().nullish(),
});

export type User = z.infer<typeof UserSchema>;
